document.getElementById('adapy-lead-form').addEventListener('submit', async function(e) {
  e.preventDefault();
  const form = e.target;
  const btn = form.querySelector('button[type="submit"]');
  btn.disabled = true;
  btn.textContent = 'Sending...';
  const data = {};
  new FormData(form).forEach((v, k) => data[k] = v);
  try {
    const res = await fetch('https://omffhncmajcazsthtccn.supabase.co/functions/v1/api-lead-submit/customquote', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    });
    const json = await res.json();
    if (json.success) {
      form.innerHTML = '<p style="color:green">' + json.message + '</p>';
      setTimeout(() => window.location = 'https://www.adapy.com', 2000);
    } else {
      alert('Error: ' + json.error);
      btn.disabled = false;
      btn.textContent = 'Submit';
    }
  } catch(err) {
    alert('Submission failed. Please try again.');
    btn.disabled = false;
    btn.textContent = 'Submit';
  }
});
</script>
